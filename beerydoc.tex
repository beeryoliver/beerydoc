\section{Commands}
\label{sec:cmds}

\subsection{Markup for package and class commands}
\label{subsec:clsmarkup}

\begin{codedoc}{\beerydocsetup}
  \begin{syntax}
    \cs{beerydocsetup} \marg{key-value list}
  \end{syntax}
  Sets the \cls{beerydoc} class keys (\S\ref{sec:keys}) in
  \meta{key-value list}.
  All assignments made by \cs{beerydocsetup} are local to the current group.
  Can be used mid-document.
\end{codedoc}

\begin{codedoc}{\key}
  \begin{syntax}
    \cs{key} \oarg{key-value list} \marg{key}
    \cs{key} \oarg{key-value list} \{\meta{key}=\meta{value}\}
    \cs{key} \oarg{key-value list} \{=\meta{value}\}
  \end{syntax}
  Prints a key, a key-value pair, or the value of a key.
  \meta{key} and \meta{value} are printed in \cs{ttfamily}.
  Within \meta{key}, line breaks are allowed after |/|.
  The |=| in \marg*{\meta{key}=\meta{value}} is reformatted and a line break is
  allowed afterwards.
  The leading |=| in \marg*{=\meta{value}} is omitted.
  Within \meta{value}, spaces are replaced by \cs{textvisiblespace}.
  Within \meta{value}, the vertical bar \texttt{\string|} may be used to
  separate each \meta{choice} when documenting a choice key.
  Sometimes the best way to typeset the value of a key is not with
  \cs{key}\marg*{=\meta{value}}, but with one of the following:
  \begin{itemize}
    \item \cs{num}, \cs{qty} (see \pkg{siunitx} package)
    \item \cs{meta}
    \item \cs{cs}
    \item verbatim shorthand \texttt{\string|}
  \end{itemize}
  Adding the optional argument sets the keys in \meta{key-value list} in path
  |key| for only that instance of \cs{key}.
\end{codedoc}

\begin{codedoc}{\cs}
  \begin{syntax}
    \cs{cs} \marg{command}
  \end{syntax}
  Prints a command.
  \meta{command} must omit the leading backslash |\|.
  The contents are printed in \cs{ttfamily}.
  \cs{meta} and other commands can be used in \meta{command}.
  The underscore |_| can be typeset directly without escaping it \cs{_}.
  When printing \pkg{expl3} commands, hyphenation is allowed after most
  underscores.
  Spaces are replaced by \cs{textvisiblespace}.
  Active vertical bars \texttt{\textbar} are replaced by that of catcode 12.
\end{codedoc}

\begin{codedoc}{\meta}
  \begin{syntax}
    \cs{meta} \marg{meta}
  \end{syntax}
  Prints a meta representation such as \meta{string}, \meta{integer}, and
  \meta{comma list}.
  The contents are printed in \cs{ttfamily} and \cs{slshape}.
  Within \meta{meta}, the underscore |_| is used to apply a subscript.
  Can be used in math mode.
\end{codedoc}

\begin{codedoc}{\marg}
  \begin{syntax}
    \cs{marg} \sarg{} \marg{argument}
  \end{syntax}
  Prints a mandatory argument.
  The contents are enclosed in braces and printed in \cs{ttfamily}.
  If the optional star is omitted, \meta{argument} is formatted using
  \cs{meta}.
\end{codedoc}

\begin{codedoc}{\oarg}
  \begin{syntax}
    \cs{oarg} \sarg{} \marg{argument}
  \end{syntax}
  Prints an optional argument.
  The contents are enclosed in square brackets and printed in \cs{ttfamily}.
  If the optional star is omitted, \meta{argument} is formatted using
  \cs{meta}.
\end{codedoc}

\begin{codedoc}{\sarg}
  Prints an optional star.
  The optional star is enclosed in angle brackets and printed in \cs{ttfamily}.
\end{codedoc}

\begin{codedoc}{\cls, \env, \pkg, \hook, \socket, \plug}
  \begin{syntax}
    \cs{cls}    \marg{class}
    \cs{pkg}    \marg{package}
    \cs{env}    \marg{environment}
    \cs{hook}   \marg{hook}
    \cs{socket} \marg{socket}
    \cs{plug}   \marg{plug}
  \end{syntax}
  Prints the name of a class, package, environment, hook, socket, or plug.
  \meta{class} and \meta{package} are printed in \cs{sffamily}.
  The rest are printed in \cs{ttfamily}.
  Within \meta{hook} and \meta{socket}, line breaks after allowed after |/|.
\end{codedoc}

\begin{codedoc}{\charcode}
  \begin{syntax}
    \cs{charcode} \oarg{integer} \marg{char}
  \end{syntax}
  Prints a character \meta{char} with catcode \meta{integer}.
  The character is printed in \cs{ttfamily} and the catcode is subscripted.
  The space is printed using \cs{textvisiblespace}.
  Some characters may need to be escaped using |\|.
  If \meta{integer} is omitted, then it equals the current catcode of
  \meta{char}.
\end{codedoc}

\subsection{Some text formatting}
\label{subsec:text}

\begin{codedoc}{\allcaps}
  \begin{syntax}
    \cs{allcaps} \oarg{integer} \marg{text}
  \end{syntax}
  Prints \meta{text} in all capital letters with added letterspacing.
  The added letterspacing equals \meta{integer} in units of thousandths of
  \unit{em} or, if the optional argument is omitted, \qty{0.05}{em}.
\end{codedoc}

\begin{codedoc}{\allsmallcaps}
  \begin{syntax}
    \cs{allsmallcaps} \sarg{} \oarg{integer} \marg{text}
  \end{syntax}
  Prints \meta{text} in all small caps with added letterspacing.
  The added letterspacing equals \meta{integer} in units of thousandths of
  \unit{em} or, if the optional argument is omitted, \qty{0.05}{em}.
  If the optional star is added, capital letters are also changed to small
  caps.
\end{codedoc}

\section{Environments}
\label{sec:envs}

This section documents the environments provided by the \cls{beerydoc} class.

\begin{codedoc}{\begin{codedoc}}
  \begin{syntax}
    |\begin{codedoc}| \oarg{key-value list} \marg{code element(s)}
    ~~\meta{code documentation}
    |\end{codedoc}|
  \end{syntax}
  Environment used to document code elements such commands, keys, and
  environments.
  \marg{code element(s)} is processed as a verbatim argument.
  Each \meta{code element} must be separated by a line break or by the value of
  the key \key{codedoc/input-separator} (\S\ref{sec:keys}).
  All leading spaces in each \meta{code element} are ignored.
  The remaining spaces are replaced by \cs{textvisiblespace}.
  (This allows the environment to document \cs{ } and command names which
  contain spaces.)
  All blank code elements are ignored.
  Adding the optional argument sets the keys in \meta{key-value list} in path
  |codedoc| for only that instance of \env{codedoc}.
  The content is contained within an \pkg{expl3} |coffin| to prevent page
  breaks.
  Cannot be used within another \env{codedoc} environment.
\end{codedoc}

\begin{codedoc}{\begin{syntax}}
  \begin{syntax}
    |\begin{syntax}|
    ~~\meta{code syntax}
    |\end{syntax}|
  \end{syntax}
  Environment used to document code syntax.
  Must be used within a \env{codedoc} environment.
  The contents are printed in \cs{ttfamily}.
  \cs{obeylines} and \cs{obeyspaces} are enabled within a \env{syntax}
  environment.
  Cannot be used within another \env{syntax} environment.
\end{codedoc}

\begin{codedoc}{\begin{codedemo}}
  \begin{syntax}
    |\begin{codedemo}|\oarg{key-value list}
    ~~\meta{code demo}
    |\end{codedemo}|
  \end{syntax}
  Environment used to demonstrate code.
  Adding the optional argument sets the keys in \meta{key-value list} in path
  |codedemo| for only that instance of \env{codedemo}.
  \meta{code demo} is first displayed verbatim in \cs{ttfamily} and \cs{small}
  and then used directly.
  The content is contained within an \pkg{expl3} |coffin| to prevent page
  breaks.
\end{codedoc}

\section{Keys}
\label{sec:keys}

\begin{codedoc}{key/neq}
  \begin{syntax}
    \key{key/neq=true|false}
  \end{syntax}
  Boolean key that sets if \cs{key} prints $\neq $ instead of $=$.
  Does not affect the behavior of \cs{key} if it is used within a \env{syntax}
  environment or in a PDF string.
  The initial value is |false|.
  The default value is |true|.
\end{codedoc}

\begin{codedoc}{codedoc/input-separator}
  \begin{syntax}
    \key{codedoc/input-separator=\meta{string}}
  \end{syntax}
  Key used in the \env{codedoc} environment.
  Sets the input separator between each \meta{code element} to \meta{string}.
  \meta{string} must not be empty or contain spaces.
  The initial value is |,|.
\end{codedoc}

\begin{codedoc}{codedemo/gobble}
  \begin{syntax}
    \key{codedemo/gobble=\meta{integer}}
  \end{syntax}
  The \env{codedemo} environment ignores an \meta{integer} number of characters
  at the start of each line.
  No characters are ignored if $\meta{integer}<1$.
  The initial value is \num{2}.
  The value is set to \num{4} within a \env{codedoc} environment.
\end{codedoc}
